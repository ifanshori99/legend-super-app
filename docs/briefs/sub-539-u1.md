```ts
// types.ts
export type SaleChannel = "dine_in" | "takeaway" | "ojol_manual";
export type OjolPlatform = "gofood" | "grabfood" | "shopeefood";
export interface Course { id: string; label: string; order: number }        // "Course 1", "Course 2", "Minuman"
export interface MenuCategory { id: string; name: string }
export interface ModifierOption { id: string; name: string; priceDeltaIdr: number }
export interface ModifierGroup { id: string; name: string; required: boolean; min: number; max: number; options: ModifierOption[] }
export interface MenuItem { id: string; name: string; categoryId: string; categoryName: string; priceIdr: number; imageUrl: string | null; soldOut: boolean; code: string | null; course: Course; modifierGroups: ModifierGroup[] }
export interface Catalog { categories: MenuCategory[]; items: MenuItem[]; updatedAt: string /*ISO*/; source: "live" | "cached" }
export interface SelectedModifier { groupId: string; optionId: string; name: string; priceDeltaIdr: number }
export type LineStatus = "draft" | "held" | "sent";
export interface CartLine { lineId: string; menuItemId: string; name: string; unitPriceIdr: number; quantity: number; modifiers: SelectedModifier[]; note: string; course: Course; hold: boolean }
export interface OrderLine extends Omit<CartLine, "hold"> { status: Exclude<LineStatus, "draft"> }
export interface PosOrderView { orderId: string; number: string; channel: SaleChannel; tableLabel: string | null; callNumber: string | null; lines: OrderLine[]; subtotalIdr: number }
export interface ParkedDraft { id: string; label: string; channel: SaleChannel; lines: CartLine[]; parkedAt: string; subtotalIdr: number }
export interface CaptainTable { id: string; label: string; zone: string; status: "occupied" | "empty"; guestCount: number; seats: number; qrSessionActive: boolean; sessionId: string | null; hasOpenCall: boolean }
export interface TableOrderView { table: CaptainTable; order: PosOrderView | null }
export type ServiceCallType = "call_waiter" | "request_bill" | "complaint";
export interface ServiceCall { id: string; tableId: string; tableLabel: string; type: ServiceCallType; reason: string; createdAt: string; status: "new" | "acknowledged" | "resolved" }
export interface OrderingConnectivity { mode: "online" | "edge_only"; catalogUpdatedAt: string | null }

// cart-logic.ts (murni, tanpa React; diuji unit)
addLine(lines, item: MenuItem, opts: { modifiers: SelectedModifier[]; note: string; quantity: number }): CartLine[]  // gabung ke baris sama bila item+modifier+note sama
setQuantity(lines, lineId, quantity): CartLine[]   // quantity <= 0 → hapus; maks 99
removeLine(lines, lineId): CartLine[]
toggleHold(lines, lineId): CartLine[]
cartTotals(lines): { itemCount: number; subtotalIdr: number; heldCount: number }  // unit = priceIdr + Σ priceDeltaIdr; integer rupiah
validateModifiers(item, modifiers): { ok: true } | { ok: false; missingGroupIds: string[] }

// draft-store.ts (useSyncExternalStore, per kunci; "kasir" atau `captain:${tableId}`; sessionStorage dengan try/catch)
useDraftCart(key: string): { lines: CartLine[]; add(item, opts): void; setQuantity(id, q): void; remove(id): void; toggleHold(id): void; clear(): void; replace(lines): void }

// hooks.ts
usePosCatalog(): UseQueryResult<Catalog>
useOrderingConnectivity(): OrderingConnectivity
useNextCallNumber(enabled: boolean): UseQueryResult<string>          // "A-12"
useSubmitOrder(): UseMutationResult<PosOrderView, Error, { channel: SaleChannel; tableLabel: string | null; ojol: { platform: OjolPlatform; externalRef: string } | null; lines: CartLine[] }>  // held lines → status held, sisanya sent; idempotency key dibuat hook
useFireLines(): UseMutationResult<PosOrderView, Error, { orderId: string; lineIds: string[] }>
useParkedDrafts(): UseQueryResult<ParkedDraft[]>
useParkDraft(): UseMutationResult<ParkedDraft, Error, { channel: SaleChannel; lines: CartLine[] }>
useResumeDraft(): UseMutationResult<ParkedDraft, Error, { id: string }>   // menghapus dari daftar
useCaptainTables(): UseQueryResult<CaptainTable[]>
useTableOrder(tableId: string): UseQueryResult<TableOrderView>
useAddToTable(): UseMutationResult<TableOrderView, Error, { tableId: string; lines: CartLine[] }>   // tambah ke sesi QR berjalan / buat pesanan
useFireTableLines(): UseMutationResult<TableOrderView, Error, { tableId: string; lineIds: string[] }>
useServiceCalls(): UseQueryResult<ServiceCall[]>        // poll 10 s
useAcknowledgeCall(): UseMutationResult<ServiceCall, Error, { id: string }>
useResolveCall(): UseMutationResult<ServiceCall, Error, { id: string }>
isContractPending(err: unknown): boolean   // re-export
```
